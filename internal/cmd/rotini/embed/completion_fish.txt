# fish completion for rotini
function __rotini_complete
    set -l tokens (commandline -opc) (commandline -ct)
    rotini __complete $tokens[2..-1] 2>/dev/null
end

function __rotini_has_results
    set -g __rotini_results (__rotini_complete)
    test (count $__rotini_results) -gt 0
end

# Offer the binary's candidates when it has any; otherwise fall back to fish's
# file completion (the binary returns nothing for path-valued flags and
# arguments, exactly so the shell takes over). Candidates arrive as
# "name<TAB>description" — fish renders that shape natively.
complete -c rotini -f -n '__rotini_has_results' -a '$__rotini_results'
complete -c rotini -F -n 'not __rotini_has_results'
