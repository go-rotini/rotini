# bash completion for rotini
# Candidates arrive as "name<TAB>description"; bash cannot render descriptions,
# so everything from the first tab is stripped.
_rotini_complete() {
    local args line IFS=$'\n'
    args=("${COMP_WORDS[@]:1:$COMP_CWORD}")
    COMPREPLY=()
    for line in $(rotini __complete "${args[@]}" 2>/dev/null); do
        COMPREPLY+=("${line%%$'\t'*}")
    done
}
complete -o default -F _rotini_complete rotini
